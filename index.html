async function signup(){ const full_name = document.getElementById("fullname").value.trim(); const phone = document.getElementById("phone").value.trim(); const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value; if(!full_name || !phone || !email || password.length < 6){ msg("أدخل جميع المعلومات وكلمة المرور من 6 أحرف على الأقل."); return; } const btn = document.querySelector('#auth button'); try { btn.disabled = true; msg("جارٍ إنشاء الحساب..."); if(!window.supabase){ throw new Error("تعذر تحميل مكتبة Supabase."); } const {data,error} = await db.auth.signUp({ email: email, password: password, options: { data: { full_name: full_name, phone: phone } } }); if(error) throw error; if(data.session && data.user){ currentUser = data.user; await loadAccount(); } else { msg("تم إنشاء الحساب بنجاح. تحقق من بريدك الإلكتروني لتأكيد الحساب، ثم سجل الدخول."); } } catch(error) { msg("خطأ: " + (error.message || "تعذر الاتصال بالخادم.")); console.error("Signup error:",error); } finally { btn.disabled = false; } }
